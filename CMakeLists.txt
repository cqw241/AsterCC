cmake_minimum_required(VERSION 3.20)
project(AsterCC VERSION 0.1.0 LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
add_executable(astercc0 stage0/compiler.cpp)
target_compile_options(astercc0 PRIVATE -Wall -Wextra -Wno-unused-parameter -Wno-unused-variable)
add_library(asterrt STATIC runtime/runtime.c)
target_compile_features(asterrt PRIVATE c_std_11)
option(ASTER_SANITIZE "Instrument seed and runtime" OFF)
if(ASTER_SANITIZE)
  foreach(t astercc0 asterrt)
    target_compile_options(${t} PRIVATE -fsanitize=address,undefined -fno-omit-frame-pointer)
    target_link_options(${t} PRIVATE -fsanitize=address,undefined)
  endforeach()
endif()
enable_testing()
find_package(Python3 REQUIRED COMPONENTS Interpreter)
add_test(NAME language COMMAND ${Python3_EXECUTABLE} ${CMAKE_SOURCE_DIR}/scripts/test.py --compiler $<TARGET_FILE:astercc0>)
set_tests_properties(language PROPERTIES WORKING_DIRECTORY ${CMAKE_SOURCE_DIR})
add_test(NAME optimization COMMAND ${Python3_EXECUTABLE} ${CMAKE_SOURCE_DIR}/scripts/test_optimizations.py)
set_tests_properties(optimization PROPERTIES WORKING_DIRECTORY ${CMAKE_SOURCE_DIR})
